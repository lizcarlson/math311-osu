\begin{exercise}
    Let $A$ be bounded below, and define $B:= \{b \in \mathbb{R} : b \text{ is a lower bound for } A\}$.  Show that $\inf(B) = \sup(A)$.
\end{exercise} 
\newpage

\begin{exercise}
    Suppose $x_n \to x$.  Prove that $y_n \to x$ where 
    \begin{align*}
        y_n := \frac{x_1 + \cdots + x_n}{n}.
    \end{align*}
 \end{exercise}
