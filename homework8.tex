\begin{exercise}
    Suppose that $f$ is differentiable at $a$.  Prove
    \begin{align*}
        \lim\limits_{h\to 0} \frac{f(a+h) - f(a-h)}{2h} = f'(a).
    \end{align*}
\end{exercise}

\begin{exercise}
    Let $f$ be a function defined on an open interval $I$ containing $a$.  Show $f'(a)$ exists if and only if there is a function $\epsilon(x)$ defined on $I$ such that
    \begin{align*}
        f(x)- f(a) = (x-a)[f'(a)-\epsilon(x)] \text{ and } \lim\limits_{x\to a} \epsilon(x) = 0.
    \end{align*}
\end{exercise}

\begin{exercise}
    Suppose $f$ is differentiable on $\mathbb{R}$ and $f(0) = 0$, $f(1) = 1$, and $f(2) = 1$.  
    \begin{enumerate}
        \item Show $f'(x) = \frac12$ for some $x \in (0,2)$.
        \item Show $f'(x) = \frac17$ for some $x \in (0,2)$.
    \end{enumerate}
\end{exercise}

\begin{exercise}
    Let $f$ be defined on $\mathbb{R}$, and suppose $|f(x) - f(y)| \leq (x-y)^2$ for all $x,y \in \mathbb{R}$.  Prove that $f$ is a constant function.
\end{exercise}

\begin{exercise}
    \begin{enumerate}
        \item Suppose $f$ is twice differentiable on an open interval $I$ and $f''(x) = 0$ for all $x \in I$.  Show $f$ has the form $f(x) = ax+b$ for suitable constants $a$ and $b$.
        \item Suppose $f$ is three times differentiable on an open interval $I$ and $f''' =0$ on $I$.  What form does $f$ have?  Prove your claim.
     \end{enumerate}
    
    
\end{exercise}
