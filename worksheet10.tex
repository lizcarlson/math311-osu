\begin{exercise}
    Prove that if an infinite series converges, then the associative property holds.  Assume $a_1 + a_2 + \cdots $ converges to a limit $L$, i.e. the sequence of partial sums $s_n := x_1 + \cdots x_n$ converges to $L$.  Show that any regrouping of the terms
    \begin{align*}
        (a_1 + \cdots a_{n_1}) + (a_{n_1+1} + \cdots + a_{n_2}) + \cdots
    \end{align*}
    leads to a series that also converges to $L$.
\end{exercise}

\begin{exercise}
    Let $x\in O$, with $O$ an open set.  Prove $a\in A$ is an isolated point of $A$ if and only if there exists an $\epsilon$-neighborhood $V_{\epsilon}(a)$, $\epsilon>0$, such that $V_{\epsilon}(a)\cap A = \emptyset$.
\end{exercise}
