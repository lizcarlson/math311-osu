\begin{exercise}
    Decide whether each conjecture is true or false.  Provide an argument for those that are true and a counterexample for each one that is false.  
    \begin{enumerate}
        \item If a derivative function is not constant, then the derivative must take on some irrational values.
        \vfill
        \item If $f'$ exists on an open interval, and there is some point $c$ where $f'(c) >0$, then there exists a $\delta$-neighborhood $V_\delta(c)$ around $c$ in which $f'(x) > 0$ for all $x \in V_{\delta}(c)$.
        \vfill
    \end{enumerate} 

    \newpage

    \textbf{Extra Problems if you have time (no grade)}

    \begin{enumerate}\setcounter{enumi}{2}
        \item  If $f$ is differentiable on an interval containing $0$ and if $\lim\limits_{x\to 0} f'(x) = L$, then it must be that $L = f'(0)$.
    \end{enumerate}
\end{exercise}
