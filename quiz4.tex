\begin{exercise}
A \textit{fixed point} of a function $f$ is a value $x$ where $f(x) = x$.  Show that if $f$ is differentiable on an interval with $f'(x) \neq 1$, then $f$ can have at most one fixed point.  (\textit{Hint: prove by contradiction.})
\end{exercise} 
\newpage

\begin{exercise}
    Recall that a function $f:A \subseteq \mathbb{R}\to \mathbb{R}$ is Lipschitz on $A$ if there exists an $L> 0$ such that for all $x,y \in A$,
    \begin{align*}
        \left| f(x)- f(y)\right| \leq L|x-y|.
    \end{align*}
    Show that if $f$ is differentiable on a closed interval $[a,b]$, and if $f'$ is continuous on $[a,b]$, then $f$ is Lipschitz on $[a,b]$.
 \end{exercise}
