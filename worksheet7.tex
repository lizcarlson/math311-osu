\begin{exercise}
    Prove the following theorem:
    \begin{theorem}
    Given $f:A \to \mathbb{R}$, $g:B\to \mathbb{R}$, assume that the range of $f$ is contained in $B$, so that the composition $g \circ f(x)= g(f(x))$ is well-defined on $A$.  If $f$ is continuous at $c\in A$ and if $g$ is continuous at $f(c)\in B$, then $g \circ f$ is continuous at $c$.
\end{theorem}
\end{exercise}

\begin{exercise}
 For every $\epsilon > 0$, find a $\delta >0$ such that
 \begin{enumerate}
     \item $f(x)= x^2$ is continuous at $x_0=1$,
     \item $f(x) = x^2$ is continuous at $x_0 = 3$, and
     \item $f(x) = x^2$ is continuous at $x_0 = 0$.
 \end{enumerate}
 What do you conjecture about the relationship between $\epsilon, \delta,$ and $x_0$?
\end{exercise}
